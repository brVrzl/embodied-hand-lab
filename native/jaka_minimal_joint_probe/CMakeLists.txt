cmake_minimum_required(VERSION 3.20)
project(jaka_minimal_joint_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SDK "${ROOT}/third_party/jaka_sdk/v2.2.7/linux")
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
  set(LIBDIR "${SDK}/python3/aarch64-linux-gnu")
else()
  set(LIBDIR "${SDK}/python3/x86_64-linux-gnu")
endif()
find_library(JAKA NAMES jakaAPI PATHS "${LIBDIR}" NO_DEFAULT_PATH REQUIRED)
add_executable(jaka_gate3c_plan_probe plan_main.cpp ../jaka_readonly_diagnostic/readonly_backend.cpp)
target_include_directories(jaka_gate3c_plan_probe PRIVATE "${SDK}/c_cpp/inc_of_c++" ../jaka_readonly_diagnostic)
target_link_libraries(jaka_gate3c_plan_probe PRIVATE "${JAKA}")
target_compile_options(jaka_gate3c_plan_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
set_target_properties(jaka_gate3c_plan_probe PROPERTIES BUILD_RPATH "${LIBDIR}")
add_executable(jaka_gate3c_motion_probe motion_main.cpp ../jaka_zero_motion_probe/zero_motion_backend.cpp)
target_include_directories(jaka_gate3c_motion_probe PRIVATE "${SDK}/c_cpp/inc_of_c++" ../jaka_zero_motion_probe)
target_link_libraries(jaka_gate3c_motion_probe PRIVATE "${JAKA}")
target_compile_options(jaka_gate3c_motion_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
set_target_properties(jaka_gate3c_motion_probe PROPERTIES BUILD_RPATH "${LIBDIR}")
add_executable(jaka_gate3c_5deg_plan_probe five_degree_plan_main.cpp ../jaka_readonly_diagnostic/readonly_backend.cpp)
target_include_directories(jaka_gate3c_5deg_plan_probe PRIVATE "${SDK}/c_cpp/inc_of_c++" ../jaka_readonly_diagnostic)
target_link_libraries(jaka_gate3c_5deg_plan_probe PRIVATE "${JAKA}")
target_compile_options(jaka_gate3c_5deg_plan_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
set_target_properties(jaka_gate3c_5deg_plan_probe PROPERTIES BUILD_RPATH "${LIBDIR}")
